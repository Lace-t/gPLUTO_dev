237 template<int dir>
238 inline void Flux (State &state, double *fx, double &pt, RunConfig &run_config)
241 {
243  double w, wg2, Bmag2, Emag2;
244  #ifdef CRG
245  double eta, eta_1, Ev, vxB;
246  #endif
247  double g, g2;
248  #ifdef GLM_MHD
249  double glm_ch = run_config.glm_ch;
250  #else
251  #endif
252 
253  constexpr int MXn = MX1 + dir;
254 
255  const double *v = state.v;
256  const double *u = state.u;
257  const double h = state.h;
258 
259  GEN_INDEX_CYCLE (dir, VX1, VXn, VXt, VXb);
260  GEN_INDEX_CYCLE (dir, BX1, BXn, BXt, BXb);
261  GEN_INDEX_CYCLE (dir, EX1, EXn, EXt, EXb);
262 
263  g = u[RHO]/v[RHO];
264  g2 = g*g;
265 
266  Bmag2 = v[BX1]*v[BX1] + v[BX2]*v[BX2] + v[BX3]*v[BX3];
267  Emag2 = v[EX1]*v[EX1] + v[EX2]*v[EX2] + v[EX3]*v[EX3];
268 
269  pt = v[PRS] + 0.5*Bmag2 + 0.5*Emag2;
270  w = v[RHO]*h;
271  wg2 = w*g2;
272 
273  fx[RHO] = u[RHO]*v[VXn];
274  fx[MX1] = wg2*v[VX1]*v[VXn] - v[BX1]*v[BXn] - v[EX1]*v[EXn];
275  fx[MX2] = wg2*v[VX2]*v[VXn] - v[BX2]*v[BXn] - v[EX2]*v[EXn];
276  fx[MX3] = wg2*v[VX3]*v[VXn] - v[BX3]*v[BXn] - v[EX3]*v[EXn];
277  fx[BXn] = 0.0;
278  fx[BXt] = -v[EXb];
279  fx[BXb] = v[EXt];
280  fx[EXn] = 0.0;
281  fx[EXt] = v[BXb];
282  fx[EXb] = -v[BXt];
283 
284  fx[ENG] = u[MXn];
285  #if RMHD_REDUCED_ENERGY
286  fx[ENG] -= fx[RHO];
287  #endif
288 
289  fx[MXn] += pt;
290 
291  #if DIVB_CONTROL == DIV_CLEANING
292  fx[BXn] = v[PSI_GLM];
293  fx[PSI_GLM] = glm_ch*glm_ch*v[BXn];
294  #endif
295 
296  #if DIVE_CONTROL == DIV_CLEANING
297  fx[EXn] = v[PHI_GLM];
298  fx[PHI_GLM] = glm_ch*glm_ch*v[EXn];
299 
300  eta = state.eta;
301  eta_1 = 1.0/eta;
302  Ev = v[EX1]*v[VX1] + v[EX2]*v[VX2] + v[EX3]*v[VX3];
303  vxB = v[VXt]*v[BXb] - v[VXb]*v[BXt];
304  fx[CRG] = g*eta_1*(v[EXn] + vxB - Ev*v[VXn]) + u[CRG]*v[VXn];
305  #endif
306 }

333 inline void PrimToCons (double *v, double *u, RunConfig &run_config)
339 {
341  double rhoh;
342  double vel2, Bmag2, Emag2;
343  double g, g2;
344  double h = Enthalpy(v, run_config);
345 
346  vel2 = v[VX1]*v[VX1] + v[VX2]*v[VX2] + v[VX3]*v[VX3];
347  g2 = 1.0/(1.0 - vel2);
348  g = sqrt(g2);
349 
350  rhoh = v[RHO]*h;
351  Bmag2 = v[BX1]*v[BX1] + v[BX2]*v[BX2] + v[BX3]*v[BX3];
352  Emag2 = v[EX1]*v[EX1] + v[EX2]*v[EX2] + v[EX3]*v[EX3];
353 
354  u[RHO] = v[RHO]*g;
355 
356  u[BX1] = v[BX1];
357  u[BX2] = v[BX2];
358  u[BX3] = v[BX3];
359

361  u[EX2] = v[EX2];
362  u[EX3] = v[EX3];
363 
364  #ifdef CRG
365  u[CRG] = v[CRG];
366  #endif
367 
368  u[MX1] = rhoh*g2*v[VX1] + v[EX2]*v[BX3] - v[EX3]*v[BX2];
369  u[MX2] = rhoh*g2*v[VX2] + v[EX3]*v[BX1] - v[EX1]*v[BX3];

376  #endif

397 inline void MaxSignalSpeed_RHD (double *v, double a2, double *cmin, double *cmax)
408 {
410  double vx, vt2, vel2, arg, sroot;
411 
412  GEN_INDEX_CYCLE (dir, VX1, VXn, VXt, VXb);
413 
414  vx = v[VXn];
415  vt2 = v[VXt]*v[VXt] + v[VXb]*v[VXb];
416  vel2 = vx*vx + vt2;
417 
418  arg = a2*(1.0 - vx*vx - vt2*a2)*(1.0 - vel2);
419  sroot = sqrt(arg);
420 
421  if (arg < 0.0) printf ("! MaxSignalSpeed_RHD(): cannot take square root\n");
422 
423  *cmax = (vx*(1.0 - a2) + sroot)/(1.0 - vel2*a2);
424  *cmin = (vx*(1.0 - a2) - sroot)/(1.0 - vel2*a2);
425 }
