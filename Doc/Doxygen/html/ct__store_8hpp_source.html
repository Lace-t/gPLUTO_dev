298

300  double Sar = Sc + fabs(Bn)/sqrR;

303  double chiR = (vr[VXn] - Sc)*(Sr - Sc)/(Sar + Sr - 2.0*Sc);
304  #elif EOS == ISOTHERMAL
305  scrh = 1.0/(Sr - Sl);
306  double rho_h = (ur[RHO]*dur - ul[RHO]*dul)*scrh;

309

311 
312  double sqrho_h = sqrt(rho_h);
313  double Sal = Sc - fabs(Bn)/sqrho_h;
314  double Sar = Sc + fabs(Bn)/sqrho_h;
315  double chiL = (vl[VXn] - Sc)*(Sl - Sc)/(Sal + Sl - 2.0*Sc);
316  double chiR = (vr[VXn] - Sc)*(Sr - Sc)/(Sar + Sr - 2.0*Sc);
317  #endif
318 
319  double nuL = (Sal + Sl)/(fabs(Sal) + fabs(Sl));
320  double nuR = (Sar + Sr)/(fabs(Sar) + fabs(Sr));
321  double nuLR = (Sal + Sar)/(fabs(Sal) + fabs(Sar));
322 
323  if ( fabs(Sar - Sal) > 1.e-9*fabs(Sr-Sl)){
324  emf->dL[dir](ti,tj,tk) = 0.5*(chiL*nuL - chiL*nuLR)
325  + 0.5*(fabs(Sal) - nuLR*Sal);
326 
327  emf->dR[dir](ti,tj,tk) = 0.5*(chiR*nuR - chiR*nuLR)
328  + 0.5*(fabs(Sar) - nuLR*Sar);
329  emf->aL[dir](ti,tj,tk) = 0.5*(1.0 + nuLR);
330  emf->aR[dir](ti,tj,tk) = 0.5*(1.0 - nuLR);
331  }else{ // HLLC, degenerate limit Bx -> 0
332  emf->dL[dir](ti,tj,tk) = 0.5*chiL*nuL + 0.5*fabs(Sal);
333  emf->dR[dir](ti,tj,tk) = 0.5*chiR*nuR + 0.5*fabs(Sar);
334 
335  emf->aL[dir](ti,tj,tk) = 0.5;
336  emf->aR[dir](ti,tj,tk) = 0.5;
337  }
338 
339  scrh = 1.0/(aR_1 - aL_1);
340 
341  // HLL Diffusion coefficients
342 
343  if (switch_to_hll){
344  emf->aL[DIRn](ti,tj,tk) = aR_1*scrh; // Right weight
345  emf->aR[DIRn](ti,tj,tk) = -aL_1*scrh; // Left weight
346  emf->dR[DIRn](ti,tj,tk) = -aL_1*aR_1*scrh; // Right diffusion coeff
347  emf->dL[DIRn](ti,tj,tk) = -aL_1*aR_1*scrh; // Left diffusion coeff
348  }
349

351  emf->e[DIRt][DIRn](ti,tj,tk) = -(aR_1*vl[VXb] - aL_1*vr[VXb])*scrh;
352 
353  emf->e_dff[DIRb][DIRn](ti,tj,tk) = 0.0;
